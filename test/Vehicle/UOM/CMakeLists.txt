cmake_minimum_required(VERSION 3.25)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(QGCUomTests LANGUAGES CXX)
    enable_testing()
    find_package(Qt6 6.8 REQUIRED COMPONENTS Core Network Test)
    add_subdirectory(../../../src/Utilities/Logging/Category LoggingCategory)
    add_subdirectory(../../../src/Utilities/Timing Timing)
    add_subdirectory(../../../src/Vehicle/UOM UomCore)
endif()
if(NOT TARGET QGCTestTiming)
    add_subdirectory(../../Utilities/Timing/Support TestTiming)
endif()
add_executable(QGCUomTest UomTest.cc)
set_target_properties(QGCUomTest PROPERTIES AUTOMOC ON)
target_link_libraries(QGCUomTest PRIVATE QGCUomCore QGCTestTiming Qt6::Test)
add_test(NAME UomTest COMMAND QGCUomTest)
set_tests_properties(UomTest PROPERTIES LABELS "Unit;Vehicle" TIMEOUT 30)

add_executable(QGCHttpUomTest HttpUomTest.cc)
set_target_properties(QGCHttpUomTest PROPERTIES AUTOMOC ON)
target_link_libraries(QGCHttpUomTest PRIVATE QGCUomCore Qt6::Test)
target_compile_definitions(QGCHttpUomTest
                           PRIVATE UOM_MOCK_SERVER="${CMAKE_CURRENT_SOURCE_DIR}/../../../tools/uom_mock_server.py"
)
add_test(NAME HttpUomTest COMMAND QGCHttpUomTest)
set_tests_properties(HttpUomTest PROPERTIES LABELS "Integration;Vehicle" TIMEOUT 30)
